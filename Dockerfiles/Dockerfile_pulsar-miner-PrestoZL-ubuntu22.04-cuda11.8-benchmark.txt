# Build with:
# docker build -t "alex88ridolfi/pulsar-miner-2.0-gpu:benchmark" -f Dockerfile_pulsar-miner-ubuntu22.04-cuda11.8-benchmark.txt .

FROM nvidia/cuda:11.8.0-devel-ubuntu22.04
LABEL maintainer="Alessandro Ridolfi <alex88.ridolfi@gmail.com>"

ENV DEBIAN_FRONTEND=noninteractive
ENV PSRHOME=/software

ARG TEMPO_SHA=aaf34c44ca1187bd92309a5b806bc865f07e731c
ARG PULSARMINER_SHA=9bec9dd79c242f64e24c3affebebb3875ced0c17

RUN apt-get update && apt-get install -y --no-install-recommends \
    git \
    ca-certificates \
    python3-pip \
    python3-dev \
    nano \
    emacs \
    less \
    latex2html \
    rsync \
    wget \
    # Tempo requirements
    autoconf \
    automake \
    libtool \
    gsl-bin \
    libgsl-dev \
    tcsh \
    gfortran \
    # PRESTO requirements
    libfftw3-bin \
    libfftw3-dev \
    libglib2.0-dev \
    libcfitsio-bin \
    libcfitsio-dev \
    pgplot5 \
    libx11-dev \
    libpng-dev \
    # PrestoZL requirements
    build-essential \
    python-is-python3 \
    && rm -rf /var/lib/apt/lists/*

RUN update-alternatives --install /usr/bin/python3 python3 /usr/bin/python3.10 1 && \
    update-alternatives --install /usr/bin/python  python  /usr/bin/python3.10 1

######################## 
# CUDA-SAMPLES (Headers only)
#######################
WORKDIR $PSRHOME
RUN git clone https://github.com/NVIDIA/cuda-samples.git && \
    cd cuda-samples && git checkout 81992093d2b8c33cab22dbf6852c070c330f1715
ENV CUDA_SAMPLES_DIR=${PSRHOME}/cuda-samples

######################## 
# TEMPO
#######################
WORKDIR $PSRHOME
RUN git clone https://git.code.sf.net/p/tempo/tempo && \
    cd tempo && \
    git checkout "${TEMPO_SHA}" && \
    test "$(git rev-parse HEAD)" = "${TEMPO_SHA}"
ENV TEMPO="${PSRHOME}/tempo"
ENV PATH="${PATH}:${PSRHOME}/tempo/bin"
WORKDIR $PSRHOME/tempo
RUN ./prepare && \
    ./configure --prefix=$PSRHOME/tempo && \
    make -j$(nproc) && \
    make install

######################## 
# PRESTO2_ON_GPU (sm_61 + sm_89)
#######################
WORKDIR $PSRHOME
RUN git clone https://github.com/alex88ridolfi/presto2_on_gpu.git && \
    cd presto2_on_gpu && \
    git checkout tags/cuda-11.6-11.8
ENV PGPLOT_DIR="/usr/lib/pgplot5"
ENV PGPLOT_FONT="/usr/lib/pgplot5/grfont.dat"
ENV PGPLOT_INCLUDES="/usr/include"
ENV PGPLOT_BACKGROUND="white"
ENV PGPLOT_FOREGROUND="black"
ENV PGPLOT_DEV="/xs"
ENV PRESTO2_ON_GPU="${PSRHOME}/presto2_on_gpu"
ENV PRESTO="${PRESTO2_ON_GPU}"
ENV PATH="${PRESTO}/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
ENV LD_LIBRARY_PATH="${PRESTO}/lib:/usr/local/cuda/lib64"
ENV LIBRARY_PATH="${PRESTO}/lib"
WORKDIR ${PRESTO2_ON_GPU}/src


RUN sed -i -E 's|CUDA_SAMPLES_PATH.*|CUDA_SAMPLES_PATH ?= /software/cuda-samples|g' Makefile
RUN sed -i '/GENCODE_FLAGS/d' Makefile && \
    sed -i '1i GENCODE_FLAGS := -gencode arch=compute_61,code=sm_61 -gencode arch=compute_61,code=compute_61 -gencode arch=compute_89,code=sm_89 -gencode arch=compute_89,code=compute_89' Makefile

RUN make makewisdom && \
    make prep && \
    make

RUN find "${PRESTO2_ON_GPU}/bin" "${PRESTO2_ON_GPU}/lib" \
      -maxdepth 1 -type f -printf '%p\n' | sort
RUN test -x "${PRESTO2_ON_GPU}/bin/accelsearch" \
      || { echo "FATAL: ${PRESTO2_ON_GPU}/bin/accelsearch missing"; \
           ls -la "${PRESTO2_ON_GPU}/bin" || true; \
           exit 1; }

RUN set -e; \
    cuda_objects=0; \
    for obj in "${PRESTO2_ON_GPU}/bin/accelsearch" "${PRESTO2_ON_GPU}"/lib/*.so; do \
      [ -f "$obj" ] || continue; \
      arches="$(cuobjdump --list-elf "$obj" 2>/dev/null || true)"; \
      if printf '%s\n' "$arches" | grep -qE 'sm_[0-9]+'; then \
        cuda_objects=$((cuda_objects + 1)); \
        echo "checking CUDA artefact: $obj"; \
        printf '%s\n' "$arches" | grep -q 'sm_61' \
          || { echo "FATAL: sm_61 missing from CUDA artefact $obj"; exit 1; }; \
        printf '%s\n' "$arches" | grep -q 'sm_89' \
          || { echo "FATAL: sm_89 missing from CUDA artefact $obj"; exit 1; }; \
      fi; \
    done; \
    [ "$cuda_objects" -gt 0 ] \
      || { echo "FATAL: no CUDA-containing presto2_on_gpu artefact found"; exit 1; }; \
    echo "presto2_on_gpu gencode OK: sm_61 sm_89"

########################
# PRESTO (Standard v4)
#######################
WORKDIR $PSRHOME
RUN git clone https://github.com/scottransom/presto.git && \
    cd presto && git checkout v4.0

ENV PRESTO="/software/presto"
ENV PATH="${PRESTO}/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
ENV LD_LIBRARY_PATH="${PRESTO}/lib:/usr/local/cuda/lib64"
ENV LIBRARY_PATH="${PRESTO}/lib"
WORKDIR ${PRESTO}/src
RUN make makewisdom
RUN make prep && make
WORKDIR ${PRESTO}

RUN python3 -m pip install --no-cache-dir \
      "numpy<2.0.0" \
      "setuptools<60" \
      meson-python \
      meson \
      ninja \
      wheel \
      matplotlib \
      scipy \
      astropy \
      pkgconfig \
      cython \
      nvidia-ml-py && \
    python3 -m pip install --no-build-isolation . && \
    python3 -c "import numpy; assert numpy.__version__.startswith('1.'), numpy.__version__" && \
    python3 -c "import presto.presto, presto.sifting; print('presto OK:', presto.__file__)"

########################
# PrestoZL (sm_61 + sm_89)
#######################
WORKDIR $PSRHOME
RUN git clone https://github.com/haeuslers/PrestoZL.git && \
    cd PrestoZL && git checkout af2e198c3f035171541bc124dc470a37e5747b0f
ENV PRESTOZL="${PSRHOME}/PrestoZL"
ENV PRESTO="${PRESTOZL}"
ENV PATH="${PRESTO}/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
ENV LD_LIBRARY_PATH="${PRESTO}/lib:/usr/local/cuda/lib64"
ENV LIBRARY_PATH="${PRESTO}/lib"

RUN if grep -qE 'sm_120|compute_120' "${PRESTOZL}/src/Makefile"; then \
      sed -i '/compute_120\|sm_120/d' "${PRESTOZL}/src/Makefile"; \
    else \
      echo "note: no sm_120/compute_120 flags found"; \
    fi
RUN ! grep -qE 'sm_120|compute_120' "${PRESTOZL}/src/Makefile" \
      || { echo "FATAL: unsupported sm_120 flags remain"; exit 1; }

RUN grep -qE '^[[:space:]]*[^[:space:]#].*compute_61.*sm_61' "${PRESTOZL}/src/Makefile" \
      || { echo "FATAL: PrestoZL Makefile has no active sm_61 rule"; exit 1; }
RUN grep -qE '^[[:space:]]*[^[:space:]#].*compute_89.*sm_89' "${PRESTOZL}/src/Makefile" \
      || { echo "FATAL: PrestoZL Makefile has no active sm_89 rule"; exit 1; }

WORKDIR ${PRESTOZL}/src
RUN make makewisdom
RUN make prep && make
RUN for arch in sm_61 sm_89; do \
      cuobjdump --list-elf "${PRESTOZL}/lib/libaccelsearch_lib.so" | grep -q "${arch}" \
        || { echo "FATAL: ${arch} missing from libaccelsearch_lib.so"; exit 1; }; \
    done && \
    echo "PrestoZL gencode OK: sm_61 sm_89"

########################
# PULSAR_MINER
#######################
WORKDIR ${PSRHOME}
RUN git clone https://github.com/haeuslers/PULSAR_MINER.git && \
    cd PULSAR_MINER && \
    git checkout "${PULSARMINER_SHA}" && \
    test "$(git rev-parse HEAD)" = "${PULSARMINER_SHA}"

# SHELL CONFIGURATION
#######################
RUN echo 'export PS1="\[\e[4;30m\]\u@\h\[\e[0m\]:\[\e[1;10m\]\w\[\e[0m\]\$ (docker)> "' >> ~/.bashrc && \
    touch ~/.bash_aliases && \
    echo 'alias ls="ls --color=auto"' >> ~/.bash_aliases && \
    echo 'alias ll="ls -l"' >> ~/.bash_aliases && \
    echo 'alias LL="ls -lL"' >> ~/.bash_aliases && \
    echo 'alias lt="ls -lrt --color"' >> ~/.bash_aliases && \
    echo 'alias lh="ls -lh --color"' >> ~/.bash_aliases && \
    echo 'alias rm="rm -i"' >> ~/.bash_aliases && \
    echo 'alias cp="cp -i"' >> ~/.bash_aliases && \
    echo 'alias mv="mv -i"' >> ~/.bash_aliases && \
    echo 'alias emacs="emacs -nw"' >> ~/.bash_aliases && \
    echo 'presto_cpu() { export PRESTO=/software/presto; export PATH=$PRESTO/bin:/usr/local/cuda/bin:/usr/bin:/bin; export LD_LIBRARY_PATH=$PRESTO/lib:/usr/local/cuda/lib64; echo "PRESTO -> $PRESTO"; }' >> ~/.bash_aliases && \
    echo 'presto_zl() { export PRESTO=/software/PrestoZL; export PATH=$PRESTO/bin:/usr/local/cuda/bin:/usr/bin:/bin; export LD_LIBRARY_PATH=$PRESTO/lib:/usr/local/cuda/lib64; echo "PRESTO -> $PRESTO"; }' >> ~/.bash_aliases && \
    echo 'presto2_gpu() { export PRESTO=/software/presto2_on_gpu; export PATH=$PRESTO/bin:/usr/local/cuda/bin:/usr/bin:/bin; export LD_LIBRARY_PATH=$PRESTO/lib:/usr/local/cuda/lib64; echo "PRESTO -> $PRESTO"; }' >> ~/.bash_aliases && \
    echo 'source ~/.bash_aliases' >> ~/.bashrc
RUN ldconfig /usr/local/lib

######################## 
# FINAL ENVIRONMENT
#######################
ENV PRESTO="${PSRHOME}/presto"
ENV PATH="${PSRHOME}/presto/bin:${PSRHOME}/PULSAR_MINER:${PSRHOME}/tempo/bin:/usr/local/cuda/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
ENV LD_LIBRARY_PATH="${PSRHOME}/presto/lib:/usr/local/cuda/lib64"
ENV LIBRARY_PATH="${PSRHOME}/presto/lib"
WORKDIR ${PSRHOME}
